class Solution {
public:
    vector<vector<int>> rotateGrid(vector<vector<int>>& grid, int k) {
        int m = grid.size();
        int n = grid[0].size();

        for(int i = 0; i < min(m, n)/2; i++) {
            int p = 2*(m-i-i) + 2*(n-i-i) - 4;
            int tk = k%p;

            vector<vector<int>> buckets(4);

            for(int x = i+1; x < n-i; x++) {
                buckets[0].push_back(grid[i][x]);
                buckets[2].push_back(grid[m-i-1][n-x-1]);
            } 
            for(int x = i+1; x < m-i; x++) {
                buckets[1].push_back(grid[x][n-1-i]);
                buckets[3].push_back(grid[m-i-x-1][i]);
            }

            for(vector<int> bucket : buckets){
                std::cout << "[ ";
                for(int num : bucket) std::cout << num << " ";
                std::cout << "]" << std::endl;
            }

        }

        return grid;
    }
};
