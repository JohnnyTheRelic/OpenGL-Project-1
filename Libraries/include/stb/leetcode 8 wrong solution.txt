class Solution {
public:
    int myAtoi(string s) {
        vector<int> nums = {};
        bool lead = true;
        bool end = false;
        bool negative = false;
        for(char c : s){
            std::cout << c << std::endl;
            if(lead){
                switch(c){
                    case ' ': break;
                    case '-': 
                        negative = true;
                        lead = false;
                        break;
                    default:
                        std::cout << "line 17 reached" << std::endl;
                        lead = false;
                        nums.push_back(single(c, end));
                        break;
                }   
            }
            else{
                nums.push_back(single(c, end));
            }
            if(end){
                break;
            }
        }
        if(nums.at(nums.size()-1)==-1) nums.pop_back();
        int sum=0;
        int place=1;
        for(int x=nums.size()-1; x<=0; x--){
            std::cout << nums.at(x) << std::endl;
            sum=sum+nums.at(x)*place;
            place=place*10;
        }
        if(negative) sum=sum*-1;
        return sum;
    }

    int single(char c, bool end){
        std::cout << c << std::endl;
        switch(c){
            case '0': return 0;
            case '1': return 1;
            case '2': return 2;
            case '3': return 3;
            case '4': return 4;
            case '5': return 5;
            case '6': return 6;
            case '7': return 7;
            case '8': return 8;
            case '9': return 9;
            default: 
                end = true;
                return -1;
        }
    }
};
